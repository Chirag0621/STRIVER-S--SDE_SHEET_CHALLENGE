class Solution {
public:
    void nextPermutation(vector<int>& nums) {
        int n = nums.size();

        int ind = -1;

        // Find breakpoint
        for(int i = n - 2; i >= 0; i--) {
            if(nums[i] < nums[i + 1]) {
                ind = i;
                break;
            }
        }

        // Last permutation
        if(ind == -1) {
            reverse(nums.begin(), nums.end());
            return;
        }

        // Find just greater element
        for(int i = n - 1; i > ind; i--) {
            if(nums[i] > nums[ind]) {
                swap(nums[i], nums[ind]);
                break;
            }
        }

        // Reverse suffix
        reverse(nums.begin() + ind + 1, nums.end());
    }
};
